\section*{Kürzeste Pfade in Graphen}

\subsection*{Topologische Sortierung $\nu$}
$(V,E)$ ist DAG. $\nu: V \to \{1,\ldots,|V|\}$ mit
$\forall (u,w)\in E: \nu(u) < \nu(w)$

\begin{itemize}
	\item Top. Sortierung ex. $\iff$ azyklisch
	\item Algorithmus hat Laufzeit $\bigO(n+m)$
		(Suche Knoten mit $\indeg=0$ und lösche diese dann samt Kanten)
\end{itemize}

\subsection*{SSSP auf DAGs}
Laufzeit: $\bigO(n+m)$

\begin{algorithm}[H]
	\ForAll{$v\in V$ (topologisch sortiert)} {
		\If{$v=s$}{
			$d(s) \gets 0$\;
			\textbf{continue}\;
		}
		$d(v) \gets \infty$\;
		\ForAll{$(u,v) \in E$}{
			\If{$d(v) > d(u) + w(u,v)$}{
				$d(v) \gets d(u) + w(u,v)$\;
				$\pi(v) \gets u$\;
			}
		}
		
	}
\end{algorithm}

\subsection*{Welchen SSSP Algorithmus?}
\begin{itemize}
	\item Keine neg. Gewichte $\implies$ Dijkstra
	\item Negative Gewichte, aber keine Zykel\\ $\implies (n-1)$-mal Bellman-Ford
	\item Negative Zykel\\ $\implies (n-1+n)$-mal Bellman-Ford
\end{itemize}

\subsection*{Dijkstra (SSSP)}
Annahme: Keine negativen Gewichte.
\begin{algorithm}[H]
	$d(s) \gets 0$\;
	\ForAll{$v\in V \setminus \{s\}$}{
		$d(v) \gets \infty$\;
	}
	$S \gets \{s\}$\;
	\While{$S \neq V$}{
		Bestimme $v \notin S$, adj. zu $S$, so dass
		$d'(v) \gets \min_{(u,v),u\in S}(d(u) + w(u,v))$ minimal ist\;
		$S \gets S \cup \{v\}$\;
		$d(v) \gets d'(v)$\;
	}
\end{algorithm}

Implementiere mit PriorityQueue (die $V \setminus S$ enthält).

Dann Laufzeit: $\bigO((|V|+|E|)\cdot\log|V|)$

\subsection*{Bellman-Ford (SSSP)}
Laufzeit: $\bigO(|V|\cdot|E|)$

\begin{algorithm}[H]
	$d(s) \gets 0$\;
	\ForAll{$v\in V \setminus \{s\}$}{
		$d(v) \gets \infty$\;
	}
	\For{$i=1$ \KwTo $|V|-1$}{
		\ForEach{$(u,v)\in E$}{
			\If{$d(v)>d(u)+w(u,v)$}{
				$d(v) \gets d(u) + w(u,v)$\;
				$\pi(v) \gets u$\;
			}
		}
	}
	\ForEach{$(u,v)\in E$}{
		\If{$d(v)>d(u)+w(u,v)$}{
			\KwRet \enquote{Negativer Zykel}\;
		}
	}
	\KwRet \enquote{Kein negativer Zykel}\;
\end{algorithm}

\subsection*{Floyd-Warshall (APSP)}
Laufzeit: $\bigO(|V|^3)$

\begin{algorithm}[H]
	$M \in \Mat(n\times n)$ mit Gewichten\;
	$D^{(0)} \gets M$\;
	\For{$k \in [1,n]$}{
		Sei $D^{(k)}\in\Mat(n\times n)$\;
		\For{$(s,t)\in[1,n]\times[1,n]$}{
			$a \gets d_{k-1}(s,t)$\;
			$b \gets d_{k-1}(s,k)+d_{k-1}(k,t)$\;
			$d_k(s,t) \gets \min\{a, b\}$\;
		}
	}
	\KwRet $D^{(n)}$\;
\end{algorithm}
