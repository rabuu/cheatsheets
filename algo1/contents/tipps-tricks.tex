\section*{Tipps \& Tricks}

\subsection*{Logarithmus-Gesetze}
\begin{itemize}
	\item $\log(a \cdot b) = \log a + \log b$
	\item $\log(a^b) = b \log a$
	\item $\log_a(b) = \frac{\log b}{\log a}$
	\item $b^{\log_b a} = a$
\end{itemize}

\subsection*{Summenformeln}
\begin{itemize}
	\item $\sum_{i=1}^n i = \frac{n\cdot(n+1)}{2}$
	\item $\sum_{i=0}^n x^i = \frac{x^{n+1} - 1}{x - 1}$
	\item $\sum_{i=0}^\infty x^i = \frac{1}{1-x}$ für $|x|<1$
	\item $\sum_{i=0}^\infty ix^i = \frac{1}{(1-x)^2}$ für $|x|<1$
\end{itemize}

\subsection*{Fakultätsabschätzung}
$\left(\frac{n}{2}\right)^{n/2} \le n! \le n^n$
