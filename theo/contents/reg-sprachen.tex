\section*{Reguläre Sprachen (REG)}
$\iff$: DEA, reg. Ausdr., reg. Grammatik
\begin{itemize}
	\item $L \text{ endlich} \implies L \in \text{REG}$
\end{itemize}

\subsection*{Abgeschlossenheit}
$\overline{L}$, $L_1 \cup L_2$, $L_1 \cap L_2$, $L_1 \setminus L_2$, $L_1 L_2$, $L^*$

\subsection*{Reguläre Ausdrücke}
$\emptyset, \e$ und $x \in \Sigma$ sind reg. Ausdrücke.

$R_1 \cup R_2$, $R_1 \circ R_2$ und $R^*$ sind reg. Ausdr.

\subsubsection*{Reg. Ausdruck $\longrightarrow$ DEA}
Konstruiere NEA:
\begin{itemize}
	\item $\cup$: Neuer Start, $\e$ zu beiden Starts
	\item $\circ$: $\e$ von Endzuständen zu Start
	\item $*$: Neuer Start, $\e$ zu alten Start, $\e$ von Endzuständen zu altem Start
\end{itemize}

\subsubsection*{DEA $\longrightarrow$ reg. Ausdruck}
$R_{ij}^{(k)}$: von $i$ zu $j$, nutze nur $1,\ldots,k$

$\displaystyle
	R_{ij}^{(0)} := \begin{cases}
		\emptyset, & i \neq j, \text{kein $\delta(i,x)=j$} \\
		\bigcup x_l, & i \neq j, \delta(i,x_l)=j \\
		\e \cup \bigcup x_l, & i = j, \delta(i, x_l) = i  \\
	\end{cases}
$

$\displaystyle
R_{ij}^{(k)} = R_{ij}^{(k-1)} \cup R_{ik}^{(k-1)} \left( R_{kk}^{(k-1)} \right)^* R_{kj}^{(k-1)}
$

\subsection*{Pumping-Lemma}
$L \in \text{REG}$. Dann:\\
$\exists k \in \N:\forall w \in L, |w| \ge k: w = xyz$,

$|xy| \le k$, $|y| > 0$, $\forall i \ge 0: xy^i z \in L$

\subsection*{Myhill-Nerode}
$\displaystyle x \sim_L y \iff \forall z\in \Sigma^*: xz \in L \Leftrightarrow yz \in L$

$L \in \text{REG} \iff |\{[u]_{\sim_L} \mid u \in \Sigma^*\}| < \infty$
