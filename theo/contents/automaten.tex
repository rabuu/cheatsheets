\section*{Endliche Automaten}

\subsection*{DEA $(Q,\Sigma,\delta,q_0,F)$}
$\delta: Q \times \Sigma \to Q$, $q_0 \in Q$, $F \subseteq Q$.

\subsubsection*{Produktautomat}
$Q = Q_1 \times Q_2$

$F$: Für Vereinigung $\lor$, für Schnitt $\land$

\subsection*{NEA $(Q, \Sigma, \delta, q_0, F)$}
$\delta : Q \times (\Sigma \cup \{\e\}) \to \mathcal{P}(Q)$ (sonst wie DEA)
\begin{itemize}
	\item NEA und DEA sind äquivalent
\end{itemize}

\subsubsection*{Potenzautomat}
$Q = \mathcal{P}(Q_1)$

Betrachte \enquote{$\e$-Einheiten}.

Starte bei Einheit für Start von $Q_1$.

Stelle Tabelle auf für jeden gebrauchten Potenzzustand und jedes $\sigma \in \Sigma$.

\subsection*{Minimalautomat}

\subsubsection*{Äquivalente Zustände eines DEAs}
$p \equiv q$: $\delta^*(p,w) \in F \iff \delta^*(q,w) \in F$

\subsubsection*{Table filling algorithm}
\begin{enumerate}
	\item Tabelle für Zustände $\{z,z'\}$ mit $z \neq z'$
	\item Markiere, wenn $z\in F$ und $z' \notin F$
	\item $\forall \sigma\in\Sigma.$ Markiere, wenn\\$\{\delta(z,\sigma),\delta(z',\sigma)\}$ markiert
	\item Wiederhole 3, bis keine Änderung
	\item $\{z,z'\}$ unmarkiert $\implies z \equiv z'$
\end{enumerate}
