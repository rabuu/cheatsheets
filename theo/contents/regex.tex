\subsection*{DEA $\longrightarrow$ reg. Ausdruck}
$R_{ij}^{(k)}$: von $i$ zu $j$, nutze nur $1,\ldots,k$

$\displaystyle
	R_{ij}^{(0)} := \begin{cases}
		\emptyset, & i \neq j, \text{kein $\delta(i,x)=j$} \\
		\bigcup x_l, & i \neq j, \delta(i,x_l)=j \\
		\e \cup \bigcup x_l, & i = j, \delta(i, x_l) = i  \\
	\end{cases}
$

$\displaystyle
R_{ij}^{(k)} = R_{ij}^{(k-1)} \cup R_{ik}^{(k-1)} \left( R_{kk}^{(k-1)} \right)^* R_{kj}^{(k-1)}
$
