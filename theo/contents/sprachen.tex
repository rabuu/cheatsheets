\section*{Formale Sprachen}

\subsection*{Reguläre Sprachen (REG)}
\begin{itemize}
	\item $L \in \text{REG} \iff \text{DEA akzeptiert $L$}$
	\item $L \in \text{REG} \iff \exists\text{regulärer Ausdruck}$
	\item $L \text{ endlich} \implies L \in \text{REG}$
	\item Abgeschlossen in REG: \\
		$\overline{L}$, $L_1 \cup L_2$, $L_1 \cap L_2$, $L_1 \setminus L_2$, $L_1 L_2$, $L^*$
\end{itemize}

\subsubsection*{Pumping-Lemma}
$L \in \text{REG}$. Dann:\\
$\exists k \in \N:\forall w \in L, |w| \ge k: w = xyz$,

mit $|xy| \le k$, $|y| > 0$, $\forall i \ge 0: xy^i z \in L$
