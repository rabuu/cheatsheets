\section*{Supervised Learning}

\subsection*{Klassifikation}

\subsubsection*{Binäre Klassifikation}
Fehler: $(FP + FN) / N$

Genauigkeit: $(TP + TN) / N$

Recall: $TP / (TP + FN)$

Präzision: $TP / (TP + FP)$

tp-Rate: $TP / (TP + FN)$ (Recall)

fp-Rate: $FP / (FP + TN)$

\subsubsection*{Bayes}
$\displaystyle P(A \mid B) = \frac{P(B \mid A) \cdot P(A)}{P(B)}$
