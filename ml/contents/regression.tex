\section*{Lineare Regression}
Modell: $g(x \mid w) = w^\top x$

Gewichte $w$ als Parametrisierung.

\subsection*{Nichtlineare Funktionen}
Basisfunktionen $\phi_j$.

$g(x \mid w) = w^\top \phi(x)$ mit $\phi(x) = [\phi_0(x), \ldots, \phi_{M-1}(x)]^\top$

\subsection*{Matrixnotation}
Designmatrix $\Phi=\mat{1 & x_1^1 & \dots & x_D^1 \\ \vdots & \vdots & \ddots & \vdots \\ 1 & x_1^N & \dots & x_D^N}$

$\hat{Y} = \Phi^\top w$

\subsection*{Lossfunktion: Squared Errors}
$\displaystyle \mathcal{L}(w) = \frac{1}{2}\sum_{i=1}^N (y_i - g(x_i \mid w))^2$

\subsubsection*{Minimierung}
$w_{\text{LS}} = (\Phi^\top \Phi)^{-1} \Phi^\top Y$,

wobei $\Phi^\top \Phi$ invertierbar ($\Phi$ voller Spaltenrang)

\subsection*{Bewertung der Modellgenauigkeit}
RSS: $\displaystyle \sum_{i=1}^N (y_i - w^\top x_i)^2$

RMSE: $\displaystyle \sqrt{\frac{1}{N}\cdot\text{RSS}(w)}$

TSS: $\displaystyle \sum_{i=1}^N (y_i - \overline{y})^2$

$\displaystyle R^2 = 1 - \frac{\text{RSS}}{\text{TSS}}$

\subsection*{Regularisierung (Ridge-Regression)}
Loss: $\displaystyle \mathcal{L}(w) + \frac{\lambda}{2}\|w\|^2$

\subsubsection*{Minimierung}
$w = (\Phi^\top \Phi + \lambda I)^{-1} \Phi^\top Y$
