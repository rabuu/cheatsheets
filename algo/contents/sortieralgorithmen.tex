\section*{Sortieralgorithmen}

\subsection*{ExpoSort}
$\Theta(2^n)$

\begin{algorithm}[H]
	\If{$n>1$}{
		ExpoSort($A$, $n-1$)\;
		\If{$A[n-1] > A[n]$}{
			swap $A[n-1]$, $A[n]$\;
		}
		ExpoSort($A$, $n-1$)
	}
\end{algorithm}

\subsection*{SelectionSort}
$\Theta(n^2)$, instable, in-place

\begin{algorithm}[H]
	\For{$i \gets 1$ \KwTo $n$}{
		$a \gets$ Minimum von $A$\;
		$B[i] \gets a$\;
		Lösche $a$ aus $A$\;
	}
	\KwRet $B$\;
\end{algorithm}

\subsection*{InsertionSort}
$\bigO(n^2)$, stable, in-place

\begin{algorithm}[H]
	\For{$j\gets 2$ \KwTo $|A|$} {
		$k \gets A[j]$\;
		$i \gets j-1$\;
		\While{$i>0$ und $A[i]>k$}{
			$A[i+1] \gets A[i]$\;
			$i \gets i-1$\;
		}
		$A[i+1] \gets k$\;
	}
\end{algorithm}

\subsection*{BubbleSort}
$\bigO(n^2)$, stable, in-place

\begin{algorithm}[H]
	\For{$i \gets 1$ \KwTo $|A| - 1$}{
		\For{$j \gets |A|$ \textbf{downto} $i+1$}{
			\If{$A[j]<A[j-1]$}{
				swap $A[j]$, $A[j-1]$\;
			}
		}
	}
\end{algorithm}

\subsection*{MergeSort}
$\Theta(n\log n)$, stable, nicht in-place

\begin{algorithm}[H]
	\eIf{$n>1$}{
		$\ell \gets$ MergeSort($A[1\ldots \lfloor\frac{n}{2}\rfloor]$)\;
		$r \gets$ MergeSort($A[\lfloor\frac{n}{2}\rfloor+1 \ldots n]$)\;
		\KwRet merge($\ell$, $r$)\;
	}{
		\KwRet $A$\;
	}
\end{algorithm}

\subsubsection*{merge($X[1\ldots k]$, $Y[1\ldots \ell]$)}
\begin{algorithm}[H]
	\If{$k=0$}{
		\KwRet $Y[1\ldots \ell]$\;
	}
	\If{$l=0$}{
		\KwRet $X[1\ldots k]$\;
	}
	\eIf{$X[1] \le Y[1]$}{
		\KwRet $X[1] \circ \text{merge}(X[2\ldots k], Y[1\ldots \ell])$\;
	}{
		\KwRet $Y[1] \circ \text{merge}(X[1\ldots k], Y[2\ldots \ell])$\;
	}
\end{algorithm}

\subsection*{QuickSort}
$\Theta(n \log n)$, instable, in-place\\
(naives Pivot, worst case $\bigO(n^2)$)

\begin{algorithm}[H]
	\If{$|A| = 1$}{
		\KwRet $A$\;
	}
	Wähle $p\in A$ als Pivot\;
	$A_{<} \gets \{a\in A \mid a < p\}$\;
	$A_{=} \gets \{a\in A \mid a = p\}$\;
	$A_{>} \gets \{a\in A \mid a > p\}$\;
	\KwRet [QSort($A_{<}$), $A_{=}$, QSort($A_{>}$)]\;
\end{algorithm}

\subsection*{Vergleichbasiertes Sortieren}
$\Omega(n \log n)$

\subsection*{CountingSort}
Annahme: Alle Zahlen in $[1,K]$

$\bigO(n+K)$, stable, nicht in-place
\begin{enumerate}
	\item Zähle Anzahl der Elemente
	\item Addiere akkumuliert auf
	\item Fülle Ausgabe rückwärts auf
\end{enumerate}

\subsection*{RadixSort}
Annahme: $d$ ist max. Anzahl der Stellen

$\bigO(d \cdot (n + K))$, stable, in-place

\begin{algorithm}[H]
	\For{$i \gets 1$ \KwTo $d$}{
		stable sort $A$ nach Stelle $i$\;
	}
\end{algorithm}
